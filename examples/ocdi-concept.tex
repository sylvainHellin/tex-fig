%% ocdi-concept.tex -- the core OCDI idea in one picture.
%%
%% One document, identified by the hash of its bytes, carries two independent
%% metadata claim sets (one per source, never merged); each set is attributed to
%% the agent that produced it and its values point at controlled-vocabulary terms.
%%
%% Requires texfig.sty. Input inside a figure* environment:
%%   \input{figures/ocdi-concept}
%% The picture is authored 25.5cm wide, so it belongs in figure*, not a single column.
%% To change the target width, \newcommand{\tffigwidth}{...} before the \input.
%% Defaults to \linewidth; standalone has no useful \linewidth, so a test wrapper
%% must set \tffigwidth itself.
\providecommand{\tffigwidth}{\linewidth}
\resizebox{\tffigwidth}{!}{%
\begin{tikzpicture}[font=]

% ===================================================================
% LEFT: the document itself
% ===================================================================
% The card is placed so that it sits vertically centred in its panel, whose
% height is set by the tallest column (the pills on the right). Its x is set so
% that the gap to the middle panel matches the one between the middle and the
% right panel, which the four edge labels there fix at 96.285pt.
\node[tfcardtitle, text=tfBlue!60!black] (dTitle) at (-1.8019,7.666)
      {ocdi:Document};
\node[tfcardsubtitle]  (dSub)  at (-1.8019,7.266) {identified by the hash of its bytes};
\node[tffield] (dF1) at (-1.8019,6.716) {ocdi:hasHash \tflit{"sha256:9f86d0..."}};
\node[tffield] (dF2) at (-1.8019,6.316) {ocdi:hasByteSize \tflit{482911}};
\node[tffield] (dF3) at (-1.8019,5.916) {ocdi:hasMediaType \tflit{"application/pdf"}};

\begin{scope}[on background layer]
  \node[tfcard, fit=(dTitle)(dSub)(dF1)(dF2)(dF3)] (cardDoc) {};
\end{scope}

\node[tfcaption] at (0.8181,1.15) {what the bytes alone determine};

% ===================================================================
% MIDDLE: two claim sets about that document
% ===================================================================
% --- upper card: the connector's claims ---------------------------
\node[tfcardtitle, text=tfGreen!42!black] (uTitle) at (8.30,10.80)
      {ocdi:Metadata};
\node[tfcardsubtitle] (uSub) at (8.30,10.40) {what the connector found};
\node[tffield] (uF1) at (8.30,9.85) {ocdi:hasFilePath};
\node[tffield] (uV1) at (8.65,9.45) {\tflit{"/LOVT/03 Plaene/A-102-P03.pdf"}};
\node[tffield] (uF2) at (8.30,8.90) {ocdi:hasDateOfRecord};
\node[tffield] (uV2) at (8.65,8.50) {\tflit{"2026-03-14T09:12Z"}};
\node[tffield] (uF3) at (8.30,7.95) {ocdi:hasPresence};
\node[tffield] (uV3) at (8.65,7.55) {\tflit{[ startedAt 2026-03-14 ]}};

\begin{scope}[on background layer]
  \node[tfcard, fit=(uTitle)(uSub)(uF1)(uV1)(uF2)(uV2)(uF3)(uV3)] (cardUp) {};
\end{scope}

% --- lower card: the parser's claims ------------------------------
\node[tfcardtitle, text=tfGreen!42!black] (lTitle) at (8.30,4.90)
      {ocdi:Metadata};
\node[tfcardsubtitle] (lSub) at (8.30,4.50) {what the parser extracted};
\node[tffield] (lF1) at (8.30,3.95) {ocdi:hasDiscipline};
\node[tffield] (lF2) at (8.30,3.55) {ocdi:hasOwner};
\node[tffield] (lF3) at (8.30,3.15) {ocdi:hasPhase};

\begin{scope}[on background layer]
  % the extra point pins the right edge to the card above, so the column aligns
  \node[tfcard, fit={(lTitle)(lSub)(lF1)(lF2)(lF3)
        ($(cardUp.east |- lF1)+(-0.317,0)$)}] (cardLow) {};
\end{scope}

\node[tfcaption] at (10.80,1.15) {one claim set per source, never merged};

% --- document to claim sets ---------------------------------------
\draw[tfarrow] ($(cardDoc.east)+(0,0.35)$) -- (cardUp.west)
  node[tfedgelabel, pos=0.52] {ocdi:hasMetadata};
\draw[tfarrow] ($(cardDoc.east)+(0,-0.35)$) -- (cardLow.west)
  node[tfedgelabel, pos=0.52] {ocdi:hasMetadata};

% --- the parser read what the connector observed ------------------
\draw[tfarrowdashed] ($(cardLow.north west)+(2.20,0)$) -- ($(cardUp.south west)+(2.20,0)$);
\node[tfedgelabel, anchor=west] at (10.33,6.20) {ocdi:isDerivedFrom};

% ===================================================================
% RIGHT: attribution (purple) and controlled vocabulary (orange)
% ===================================================================
% The column is set far enough right for the four edge labels to sit in open
% space between the middle and right panels. Each pill is pinned to the card it
% belongs to: the connector to the middle of the upper card, the discipline to
% the middle of the lower card, so both those arrows run exactly horizontal. The
% parser and the owner sit 1.40cm above and below the discipline, so their two
% arrows leave the lower card at mirrored angles.
\node[tfpill={tfPurple}{4.70cm}] (srcConn) at (20.20,0 |- cardUp.east)
      {OneDrive connector\\[1pt]{\tiny prov:SoftwareAgent}};
\node[tfpill={tfPurple}{4.70cm}] (srcPars) at ($(20.20,0 |- cardLow.east)+(0,1.40)$)
      {Basenord parser v3\\[1pt]{\tiny prov:SoftwareAgent}};
\node[tfpill={tfOrange}{4.70cm}] (vocDisc) at (20.20,0 |- cardLow.east)
      {inst:discipline-architecture\\[1pt]{\tiny ocdi:Discipline}};
\node[tfpill={tfOrange}{4.70cm}] (vocOwn) at ($(20.20,0 |- cardLow.east)+(0,-1.40)$)
      {Hines\\[1pt]{\tiny ocdi:Organization}};

\draw[tfarrowc=tfPurple] (cardUp.east) -- (srcConn.west)
  node[tfedgelabelc=tfPurple, pos=0.50] {ocdi:hasSource};
% the three arrows of the lower card all leave the same point on its east edge,
% so the outer pair reads as one symmetric fan around the horizontal middle one
\draw[tfarrowc=tfPurple] (cardLow.east) -- (srcPars.west)
  node[tfedgelabelc=tfPurple, pos=0.50] {ocdi:hasSource};
\draw[tfarrowc=tfOrange] (cardLow.east) -- (vocDisc.west)
  node[tfedgelabelc=tfOrange, pos=0.50] {ocdi:hasDiscipline};
\draw[tfarrowc=tfOrange] (cardLow.east) -- (vocOwn.west)
  node[tfedgelabelc=tfOrange, pos=0.50] {ocdi:hasOwner};

\node[tfcaption] at (20.20,1.15)
      {every claim attributed,\\every value from a controlled vocabulary};

% ===================================================================
% Background panels (one per column of the story)
% ===================================================================
% Fitted around the column contents, so every panel carries the same inner
% padding (set once in tfpanel). The extra coordinates share the x of the
% column they belong to and only pin the y: every panel is given the topmost and
% the bottommost edge of the whole picture (the upper card above, the owner pill
% below), so the three share one top and one bottom edge.
\begin{pgfonlayer}{grouping}
  \node[tfpanel, fit={(cardDoc)
        (cardDoc.west |- cardUp.north) (cardDoc.west |- srcConn.north)
        (cardDoc.west |- cardLow.south) (cardDoc.west |- vocOwn.south)}] (panL) {};
  \node[tfpanel, fit={(cardUp) (cardLow)
        (cardUp.west |- srcConn.north) (cardUp.west |- vocOwn.south)}] (panM) {};
  \node[tfpanel, fit={(srcConn) (srcPars) (vocDisc) (vocOwn)
        (srcConn.west |- cardUp.north) (srcConn.west |- cardLow.south)}] (panR) {};
\end{pgfonlayer}

\end{tikzpicture}}
